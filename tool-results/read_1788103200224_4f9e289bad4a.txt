    55→    55→/** Hold-to-unlock button with a 1.2s progress ring. */
    56→    56→function LockOverlay({ onUnlock }: { onUnlock: () => void }) {
    57→    57→  const [holding, setHolding] = useState(false)
    58→    58→  const timerRef = useRef<number | null>(null)
    59→    59→  const C = 2 * Math.PI * 26
    60→    60→
    61→    61→  const start = () => {
    62→    62→    setHolding(true)
    63→    63→    if (timerRef.current) window.clearTimeout(timerRef.current)
    64→    64→    timerRef.current = window.setTimeout(onUnlock, 1200)
    65→    65→  }
    66→    66→  const cancel = () => {
    67→    67→    setHolding(false)
    68→    68→    if (timerRef.current) {
    69→    69→      window.clearTimeout(timerRef.current)
    70→    70→      timerRef.current = null
    71→    71→    }
    72→    72→  }
    73→    73→  useEffect(
    74→    74→    () => () => {
    75→    75→      if (timerRef.current) window.clearTimeout(timerRef.current)
    76→    76→    },
    77→    77→    []
    78→    78→  )
    79→    79→
    80→    80→  return (
    81→    81→    <div className="absolute inset-x-0 top-0 z-30 flex flex-col items-center gap-4 pt-5">
    82→    82→      <motion.div
    83→    83→        initial={{ opacity: 0, y: -8 }}
    84→    84→        animate={{ opacity: 1, y: 0 }}
    85→    85→        transition={{ duration: 0.2 }}
    86→    86→        className="flex items-center gap-2 rounded-full border border-white/10 bg-black/60 px-4 py-2 backdrop-blur"
    87→    87→      >
    88→    88→        <Lock className="h-3.5 w-3.5 text-white/80" />
    89→    89→        <span className="text-xs text-white/80">Player locked — hold to unlock</span>
    90→    90→      </motion.div>
    91→    91→      <button
    92→    92→        aria-label="Hold 1.2 seconds to unlock"
    93→    93→        onPointerDown={start}
    94→    94→        onPointerUp={cancel}
    95→    95→        onPointerLeave={cancel}
    96→    96→        onPointerCancel={cancel}
    97→    97→        onContextMenu={(e) => e.preventDefault()}
    98→    98→        className="relative flex h-14 w-14 touch-none items-center justify-center rounded-full border border-white/15 bg-black/60 text-white/90 backdrop-blur active:scale-95"
    99→    99→      >
   100→   100→        <svg viewBox="0 0 56 56" className="absolute inset-0 h-full w-full -rotate-90">
   101→   101→          <circle cx="28" cy="28" r="26" fill="none" stroke="rgba(255,255,255,0.12)" strokeWidth="3" />
   102→   102→          <circle
   103→   103→            cx="28"
   104→   104→            cy="28"
   105→   105→            r="26"
   106→   106→            fill="none"
   107→   107→            stroke="var(--vx-accent)"
   108→   108→            strokeWidth="3"
   109→   109→            strokeLinecap="round"
   110→   110→            strokeDasharray={C}
   111→   111→            strokeDashoffset={holding ? 0 : C}
   112→   112→            style={{ transition: holding ? 'stroke-dashoffset 1.2s linear' : 'stroke-dashoffset 0.25s ease' }}
   113→   113→          />
   114→   114→        </svg>
   115→   115→        <LockOpen className="h-5 w-5" />
   116→   116→      </button>
   117→   117→    </div>
   118→   118→  )
   119→   119→}
